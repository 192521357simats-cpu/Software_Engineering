pipeline {
    agent any

    stages {

        stage('Build') {
            steps {
                bat 'docker build -t flask-api .'
            }
        }

        stage('Test') {
            steps {
                bat 'docker images'
            }
        }

        stage('Deploy') {
            steps {
                bat 'docker run -d -p 5000:5000 flask-api'
            }
        }

    }
}
